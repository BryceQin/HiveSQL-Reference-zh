# 1. 数值类型（Numeric Types）
## 1.1 `TINYINT`
- 1 字节有符号整数，从`-128`到`127`。<br>

```sql
-- 建表时指定 TINYINT 类型
CREATE TABLE tinyint_demo (
    id INT,
    score TINYINT COMMENT '评分，范围 -128 到 127'
);

-- 插入数据
INSERT INTO tinyint_demo VALUES (1, 100), (2, -50), (3, 127);

-- 查询
SELECT * FROM tinyint_demo;
-- +-----+--------+
-- | id  | score  |
-- +-----+--------+
-- | 1   | 100    |
-- | 2   | -50    |
-- | 3   | 127    |
-- +-----+--------+

-- 使用后缀 Y 表示 TINYINT 字面量
SELECT 100Y;
-- 返回 100 (TINYINT 类型)
```

## 1.2 `SMALLINT`
- 2 字节有符号整数，从`-32,768`到`32,767`。<br>

```sql
-- 建表时指定 SMALLINT 类型
CREATE TABLE smallint_demo (
    id INT,
    quantity SMALLINT COMMENT '数量，范围 -32768 到 32767'
);

-- 插入数据
INSERT INTO smallint_demo VALUES (1, 15000), (2, -10000), (3, 32767);

-- 查询
SELECT * FROM smallint_demo;
-- +-----+-----------+
-- | id  | quantity  |
-- +-----+-----------+
-- | 1   | 15000     |
-- | 2   | -10000    |
-- | 3   | 32767     |
-- +-----+-----------+

-- 使用后缀 S 表示 SMALLINT 字面量
SELECT 1000S;
-- 返回 1000 (SMALLINT 类型)
```

## 1.3 **`INT/INTEGER`**
- 4 字节有符号整数，从`-2,147,483,648`到`2,147,483,647`。<br>
- 在2.20版本中，引入`INTEGER`作为`INT`的同义词。<br>
- 整数默认为`INT`,除非超过了`INT`的最大值，才会转为`BIGINT`，或者带有以下后缀。<br>
    - `Y`：`TINYINT`<br>
    - `S`：`SMALLINT`<br>
    - `L`：`BIGINT`<br>

```sql
-- 建表时指定 INT 类型
CREATE TABLE int_demo (
    id INT COMMENT '主键',
    user_id INTEGER COMMENT '用户ID，INTEGER 与 INT 等效',
    age INT COMMENT '年龄'
);

-- 插入数据
INSERT INTO int_demo VALUES (1, 10001, 25), (2, 10002, 30);

-- 查询
SELECT * FROM int_demo;
-- +-----+----------+------+
-- | id  | user_id  | age  |
-- +-----+----------+------+
-- | 1   | 10001    | 25   |
-- | 2   | 10002    | 30   |
-- +-----+----------+------+

-- 整数后缀演示
SELECT TYPEOF(100), TYPEOF(100Y), TYPEOF(100S), TYPEOF(100L);
-- 分别返回: INT, TINYINT, SMALLINT, BIGINT
```

## 1.4 **BIGINT**
- 8 字节有符号整数，从`-9,223,372,036,854,775,808`到`9,223,372,036,854,775,807`。<br>

```sql
-- 建表时指定 BIGINT 类型
CREATE TABLE bigint_demo (
    id BIGINT COMMENT '大整数主键',
    total_visits BIGINT COMMENT '总访问量'
);

-- 插入数据（使用后缀 L 表示 BIGINT 字面量）
INSERT INTO bigint_demo VALUES
    (1, 9223372036854775807L),
    (2, -9223372036854775808L);

-- 查询
SELECT id, total_visits FROM bigint_demo;
-- +-----+----------------------+
-- | id  | total_visits         |
-- +-----+----------------------+
-- | 1   | 9223372036854775807  |
-- | 2   | -9223372036854775808 |
-- +-----+----------------------+

-- 超出 INT 范围的字面量自动转为 BIGINT
SELECT TYPEOF(2147483648);
-- 返回 BIGINT（因为 2147483648 > INT 最大值 2147483647）
```

## 1.5 `FLOAT`
- 4 字节单精度浮点数。<br>

```sql
-- 建表时指定 FLOAT 类型
CREATE TABLE float_demo (
    id INT,
    temperature FLOAT COMMENT '温度',
    percentage FLOAT COMMENT '百分比'
);

-- 插入数据
INSERT INTO float_demo VALUES
    (1, 36.5, 0.85),
    (2, -10.2, 0.333),
    (3, CAST(1.23456789 AS FLOAT), 0.0);

-- 查询（注意 FLOAT 精度有限，可能显示为近似值）
SELECT * FROM float_demo;
-- +-----+--------------+-------------+
-- | id  | temperature  | percentage  |
-- +-----+--------------+-------------+
-- | 1   | 36.5         | 0.85        |
-- | 2   | -10.2        | 0.333       |
-- | 3   | 1.2345679    | 0.0         |
-- +-----+--------------+-------------+

-- 科学计数法表示 FLOAT
SELECT CAST(1.5e3 AS FLOAT);
-- 返回 1500.0
```

## 1.6 `DOUBLE/DOUBLE PRECISION`
- 8 字节双精度浮点数。<br>
- `DOUBLE PRECISION` 是 `DOUBLE` 的别名，自2.2.0版本开始使用。<br>

```sql
-- 建表时指定 DOUBLE 类型
CREATE TABLE double_demo (
    id INT,
    precise_value DOUBLE COMMENT '高精度浮点数',
    scientific_val DOUBLE PRECISION COMMENT 'DOUBLE PRECISION 与 DOUBLE 等效'
);

-- 插入数据（Hive 中浮点数字面量默认为 DOUBLE）
INSERT INTO double_demo VALUES
    (1, 3.141592653589793, 2.718281828459045),
    (2, -1.23e10, 6.626e-34);

-- 查询
SELECT * FROM double_demo;
-- +-----+---------------------+---------------------+
-- | id  | precise_value       | scientific_val      |
-- +-----+---------------------+---------------------+
-- | 1   | 3.141592653589793   | 2.718281828459045   |
-- | 2   | -1.23E10            | 6.626E-34           |
-- +-----+---------------------+---------------------+

-- 验证 DOUBLE 与 DOUBLE PRECISION 等效
SELECT TYPEOF(CAST(1.0 AS DOUBLE)), TYPEOF(CAST(1.0 AS DOUBLE PRECISION));
-- 两列均返回 DOUBLE
```

## 1.7 **`DECIMAL/NUMERIC`**
- 0.11.0版本中引入，精度为 38 位。<br>
- 0.13.0版本引入了用户可定义的精度。<br>
- `NUMERIC`与`DECIMAL`是相同，自3.0.0版本开始使用。<br>
- 使用`DECIMAL(precision, scale) `定义，`precision`表示总位数，`scale`表示小数位数。如果未指定precision，则默认为10。如果未指定scale，它将默认为0（无小数位）。<br>
- 如果超出总位数会为`NULL`，如果超出精度会四舍五入。<br>

```sql
-- 建表时指定 DECIMAL 类型
CREATE TABLE decimal_demo (
    id INT,
    price DECIMAL(10, 2) COMMENT '价格，总位数10，小数位2',
    amount DECIMAL(18, 6) COMMENT '金额，总位数18，小数位6',
    quantity DECIMAL COMMENT '默认为 DECIMAL(10, 0)，无小数位'
);

-- 插入数据
INSERT INTO decimal_demo VALUES
    (1, CAST(99999999.99 AS DECIMAL(10, 2)), CAST(123456789012.123456 AS DECIMAL(18, 6)), 100),
    (2, CAST(0.01 AS DECIMAL(10, 2)), CAST(-0.000001 AS DECIMAL(18, 6)), 50);

-- 查询
SELECT * FROM decimal_demo;
-- +-----+-------------+-----------------------+-----------+
-- | id  | price       | amount                | quantity  |
-- +-----+-------------+-----------------------+-----------+
-- | 1   | 99999999.99 | 123456789012.123456   | 100       |
-- | 2   | 0.01        | -0.000001             | 50        |
-- +-----+-------------+-----------------------+-----------+

-- 超出小数位的值会四舍五入
SELECT CAST(123.456 AS DECIMAL(5, 2));
-- 返回 123.46

-- 超出总位数会返回 NULL
SELECT CAST(12345.67 AS DECIMAL(5, 2));
-- 返回 NULL（总位数6超出了precision=5）

-- NUMERIC 与 DECIMAL 等效
SELECT TYPEOF(CAST(1.0 AS DECIMAL(5,2))), TYPEOF(CAST(1.0 AS NUMERIC(5,2)));
-- 两列均返回 decimal(5,2)
```

# 2. 日期时间类型（Date/Time Types）
## 2.1 **`TIMESTAMP`**
- 从0.8.0版本开始使用。<br>
- 格式：`yyyy-mm-dd hh:mm:ss[.f...]` <br>

```sql
-- 建表时指定 TIMESTAMP 类型
CREATE TABLE timestamp_demo (
    id INT,
    event_time TIMESTAMP COMMENT '事件时间',
    log_time TIMESTAMP COMMENT '日志时间'
);

-- 插入数据
INSERT INTO timestamp_demo VALUES
    (1, '2024-01-15 10:30:00', '2024-01-15 10:30:00.123'),
    (2, '2024-06-01 08:00:00', CAST('2024-06-01 08:00:00' AS TIMESTAMP)),
    (3, TIMESTAMP '2024-12-31 23:59:59', NULL);

-- 查询
SELECT * FROM timestamp_demo;
-- +-----+------------------------+------------------------+
-- | id  | event_time             | log_time               |
-- +-----+------------------------+------------------------+
-- | 1   | 2024-01-15 10:30:00.0  | 2024-01-15 10:30:00.123|
-- | 2   | 2024-06-01 08:00:00.0  | 2024-06-01 08:00:00.0  |
-- | 3   | 2024-12-31 23:59:59.0  | NULL                   |
-- +-----+------------------------+------------------------+

-- 使用内置函数转换 TIMESTAMP
SELECT
    CURRENT_TIMESTAMP() AS now_ts,
    UNIX_TIMESTAMP('2024-01-15 10:30:00') AS unix_ts,
    FROM_UNIXTIME(1705312200) AS from_unix;
```

## 2.2 **`DATE`**
- 从0.12.0版本开始使用。<br>
- 格式：`yyyy-MM-dd`，范围：`0000-01-01`到`9999-12-31`。<br>

```sql
-- 建表时指定 DATE 类型
CREATE TABLE date_demo (
    id INT,
    birth_date DATE COMMENT '出生日期',
    order_date DATE COMMENT '下单日期'
);

-- 插入数据（必须使用 yyyy-MM-dd 格式）
INSERT INTO date_demo VALUES
    (1, DATE '1990-05-20', '2024-01-15'),
    (2, DATE '2000-12-01', '2024-06-01'),
    (3, CAST('1995-08-08' AS DATE), NULL);

-- 查询
SELECT * FROM date_demo;
-- +-----+-------------+-------------+
-- | id  | birth_date  | order_date  |
-- +-----+-------------+-------------+
-- | 1   | 1990-05-20  | 2024-01-15  |
-- | 2   | 2000-12-01  | 2024-06-01  |
-- | 3   | 1995-08-08  | NULL        |
-- +-----+-------------+-------------+

-- DATE 范围边界测试
SELECT DATE '0000-01-01' AS min_date, DATE '9999-12-31' AS max_date;
-- +-------------+-------------+
-- | min_date    | max_date    |
-- +-------------+-------------+
-- | 0000-01-01  | 9999-12-31  |
-- +-------------+-------------+

-- 常用日期函数
SELECT
    CURRENT_DATE() AS today,
    DATE_ADD(DATE '2024-01-15', 30) AS after_30_days,
    DATE_SUB(DATE '2024-01-15', 30) AS before_30_days,
    DATEDIFF(DATE '2024-06-01', DATE '2024-01-15') AS diff_days,
    YEAR(DATE '2024-01-15') AS year_val,
    MONTH(DATE '2024-01-15') AS month_val,
    DAY(DATE '2024-01-15') AS day_val;
-- +-------------+---------------+---------------+---------------+------------+------------+----------+
-- | today       | after_30_days | before_30_days| diff_days     | year_val   | month_val  | day_val  |
-- +-------------+---------------+---------------+---------------+------------+------------+----------+
-- | 2024-01-15  | 2024-02-14    | 2023-12-16    | 138           | 2024       | 1          | 15       |
-- +-------------+---------------+---------------+---------------+------------+------------+----------+

-- 从 TIMESTAMP 转换为 DATE
SELECT CAST(TIMESTAMP '2024-01-15 10:30:00' AS DATE);
-- 返回 2024-01-15
```

## 2.3 `INTERVAL`
- 从1.2.0版本开始使用。<br>

```sql
-- INTERVAL 用于日期时间的加减运算
-- 语法: date/timestamp +/- INTERVAL <value> <unit>

-- ========== Year-Month 间隔（year-month interval） ==========
-- 支持单位: YEAR, MONTH (以及复合形式 YEAR TO MONTH)

-- 加/减年
SELECT
    DATE '2024-01-15' + INTERVAL 1 YEAR AS add_year,
    DATE '2024-01-15' - INTERVAL 2 YEAR AS sub_year;
-- +-------------+-------------+
-- | add_year    | sub_year    |
-- +-------------+-------------+
-- | 2025-01-15  | 2022-01-15  |
-- +-------------+-------------+

-- 加/减月
SELECT
    DATE '2024-01-15' + INTERVAL 3 MONTH AS add_month,
    DATE '2024-01-15' - INTERVAL 1 MONTH AS sub_month;
-- +-------------+-------------+
-- | add_month   | sub_month   |
-- +-------------+-------------+
-- | 2024-04-15  | 2023-12-15  |
-- +-------------+-------------+

-- TIMESTAMP 的 year-month 间隔
SELECT
    TIMESTAMP '2024-01-15 10:30:00' + INTERVAL 1 YEAR AS ts_add_year;
-- +------------------------+
-- | ts_add_year            |
-- +------------------------+
-- | 2025-01-15 10:30:00.0  |
-- +------------------------+

-- ========== Day-Time 间隔（day-time interval） ==========
-- 支持单位: DAY, HOUR, MINUTE, SECOND (以及复合形式 DAY TO SECOND 等)

-- 加/减天
SELECT
    DATE '2024-01-15' + INTERVAL 10 DAY AS add_day,
    TIMESTAMP '2024-01-15 10:30:00' - INTERVAL 5 DAY AS sub_day_ts;
-- +-------------+------------------------+
-- | add_day     | sub_day_ts             |
-- +-------------+------------------------+
-- | 2024-01-25  | 2024-01-10 10:30:00.0  |
-- +-------------+------------------------+

-- 加/减小时
SELECT
    TIMESTAMP '2024-01-15 10:30:00' + INTERVAL 2 HOUR AS add_hour,
    TIMESTAMP '2024-01-15 10:30:00' - INTERVAL 1 HOUR AS sub_hour;
-- +------------------------+------------------------+
-- | add_hour               | sub_hour               |
-- +------------------------+------------------------+
-- | 2024-01-15 12:30:00.0  | 2024-01-15 09:30:00.0  |
-- +------------------------+------------------------+

-- 加/减分钟
SELECT TIMESTAMP '2024-01-15 10:30:00' + INTERVAL 45 MINUTE AS add_minute;
-- +------------------------+
-- | add_minute             |
-- +------------------------+
-- | 2024-01-15 11:15:00.0  |
-- +------------------------+

-- 加/减秒
SELECT TIMESTAMP '2024-01-15 10:30:00' + INTERVAL 30 SECOND AS add_second;
-- +------------------------+
-- | add_second             |
-- +------------------------+
-- | 2024-01-15 10:30:30.0  |
-- +------------------------+

-- 组合使用多个 INTERVAL
SELECT
    TIMESTAMP '2024-01-15 10:30:00'
        + INTERVAL 1 YEAR
        + INTERVAL 2 MONTH
        + INTERVAL 3 DAY
        + INTERVAL 4 HOUR AS combined_result;
-- +------------------------+
-- | combined_result        |
-- +------------------------+
-- | 2025-03-18 14:30:00.0  |
-- +------------------------+
```

# 3. 字符串类型（String Types）
## 3.1 **`STRING`**
- 以用单引号(')或双引号(")表示。<br>

```sql
-- 建表时指定 STRING 类型
CREATE TABLE string_demo (
    id INT,
    name STRING COMMENT '用户名',
    description STRING COMMENT '描述信息'
);

-- 插入数据（支持单引号和双引号）
INSERT INTO string_demo VALUES
    (1, 'Alice', "普通用户"),
    (2, 'Bob', 'VIP用户'),
    (3, "Charlie", "管理员");

-- 查询
SELECT * FROM string_demo;
-- +-----+----------+--------------+
-- | id  | name     | description  |
-- +-----+----------+--------------+
-- | 1   | Alice    | 普通用户     |
-- | 2   | Bob      | VIP用户      |
-- | 3   | Charlie  | 管理员       |
-- +-----+----------+--------------+

-- 字符串常用函数
SELECT
    CONCAT('Hello', ' ', 'World') AS concat_example,
    LENGTH('Hello World') AS str_length,
    UPPER('hello') AS upper_case,
    LOWER('HELLO') AS lower_case,
    SUBSTR('Hello World', 1, 5) AS substring_example,
    TRIM('  hello  ') AS trim_example,
    REPLACE('Hello World', 'World', 'Hive') AS replace_example;
-- +-----------------+-------------+-------------+-------------+--------------------+--------------+----------------+
-- | concat_example  | str_length  | upper_case  | lower_case  | substring_example  | trim_example | replace_example|
-- +-----------------+-------------+-------------+-------------+--------------------+--------------+----------------+
-- | Hello World     | 11          | HELLO       | hello       | Hello              | hello        | Hello Hive     |
-- +-----------------+-------------+-------------+-------------+--------------------+--------------+----------------+

-- STRING 可以存储任意长度的文本，无长度限制
SELECT LENGTH(CAST('' AS STRING)) AS empty_length;
-- 返回 0
```

## 3.2 `VARCHAR`
- 从0.12.0版本开始使用。<br>
- 最大长度65535，如果要转换/分配给`varchar`的字符串值超出了长度说明符，则该字符串将被静默截断。<br>

```sql
-- 建表时指定 VARCHAR 类型（必须指定最大长度）
CREATE TABLE varchar_demo (
    id INT,
    username VARCHAR(50) COMMENT '用户名，最长50字符',
    email VARCHAR(255) COMMENT '邮箱，最长255字符'
);

-- 插入数据
INSERT INTO varchar_demo VALUES
    (1, 'alice_wang', 'alice@example.com'),
    (2, 'bob_zhang', 'bob@example.com');

-- 查询
SELECT * FROM varchar_demo;
-- +-----+--------------+--------------------+
-- | id  | username     | email              |
-- +-----+--------------+--------------------+
-- | 1   | alice_wang   | alice@example.com  |
-- | 2   | bob_zhang    | bob@example.com    |
-- +-----+--------------+--------------------+

-- 超出长度时静默截断（不会报错）
SELECT CAST('ABCDEFGHIJKLMNOPQRSTUVWXYZ' AS VARCHAR(10));
-- 返回 'ABCDEFGHIJ'（被截断为10个字符）

-- VARCHAR 最大长度为 65535
CREATE TABLE varchar_max_demo (
    id INT,
    content VARCHAR(65535) COMMENT '最大长度的 VARCHAR'
);
```

## 3.3 `CHAR`
- 从0.13.0版本开始使用。<br>
- 最大长度255，类似于`VARCHAR`，但它们是固定长度的，这意味着比指定长度值短的值用空格填充。<br>

```sql
-- 建表时指定 CHAR 类型（必须指定长度）
CREATE TABLE char_demo (
    id INT,
    country_code CHAR(3) COMMENT '国家代码，固定3字符',
    status_flag CHAR(1) COMMENT '状态标记，固定1字符'
);

-- 插入数据（不足长度时会用空格填充）
INSERT INTO char_demo VALUES
    (1, 'CN', 'Y'),
    (2, 'US', 'N'),
    (3, 'JP', 'Y');

-- 查询
SELECT * FROM char_demo;
-- +-----+---------------+--------------+
-- | id  | country_code  | status_flag  |
-- +-----+---------------+--------------+
-- | 1   | CN            | Y            |
-- | 2   | US            | N            |
-- | 3   | JP            | Y            |
-- +-----+---------------+--------------+

-- CHAR 最大长度为 255
CREATE TABLE char_max_demo (
    id INT,
    code CHAR(255) COMMENT '最大长度的 CHAR'
);

-- CHAR 与 VARCHAR 的区别
-- CHAR(5) 存储 'AB' 时实际存储为 'AB   '（右侧填充3个空格）
-- VARCHAR(5) 存储 'AB' 时实际存储为 'AB'（不填充空格）
SELECT
    LENGTH(CAST('AB' AS CHAR(5))) AS char_length,
    LENGTH(CAST('AB' AS VARCHAR(5))) AS varchar_length;
-- +--------------+-----------------+
-- | char_length  | varchar_length  |
-- +--------------+-----------------+
-- | 5            | 2               |
-- +--------------+-----------------+

-- 比较时尾部空格会被忽略（遵循 SQL 标准）
SELECT CAST('AB' AS CHAR(5)) = CAST('AB' AS VARCHAR(2)) AS compare_result;
-- 返回 true
```

# 4. 杂项类型（Misc Types）
## 4.1 `BOOLEAN`
- 布尔类型，取值为`TRUE`或`FALSE`。<br>

```sql
-- 建表时指定 BOOLEAN 类型
CREATE TABLE boolean_demo (
    id INT,
    is_active BOOLEAN COMMENT '是否激活',
    is_deleted BOOLEAN COMMENT '是否已删除',
    has_permission BOOLEAN COMMENT '是否有权限'
);

-- 插入数据
INSERT INTO boolean_demo VALUES
    (1, TRUE, FALSE, TRUE),
    (2, FALSE, FALSE, TRUE),
    (3, TRUE, TRUE, FALSE);

-- 查询
SELECT * FROM boolean_demo;
-- +-----+------------+-------------+-----------------+
-- | id  | is_active  | is_deleted  | has_permission  |
-- +-----+------------+-------------+-----------------+
-- | 1   | true       | false       | true            |
-- | 2   | false      | false       | true            |
-- | 3   | true       | true        | false           |
-- +-----+------------+-------------+-----------------+

-- BOOLEAN 字面量
SELECT TRUE, FALSE;
-- 也可以用 'true'/'false' 字符串转换
SELECT CAST('true' AS BOOLEAN), CAST('false' AS BOOLEAN);

-- BOOLEAN 在 WHERE 条件中的应用
SELECT * FROM boolean_demo WHERE is_active = TRUE;
-- +-----+------------+-------------+-----------------+
-- | id  | is_active  | is_deleted  | has_permission  |
-- +-----+------------+-------------+-----------------+
-- | 1   | true       | false       | true            |
-- | 3   | true       | true        | false           |
-- +-----+------------+-------------+-----------------+

-- BOOLEAN 逻辑运算
SELECT
    TRUE AND FALSE AS and_result,
    TRUE OR FALSE AS or_result,
    NOT TRUE AS not_result,
    TRUE AND NOT FALSE AS combined_result;
-- +-------------+------------+-------------+------------------+
-- | and_result  | or_result  | not_result  | combined_result  |
-- +-------------+------------+-------------+------------------+
-- | false       | true       | false       | true             |
-- +-------------+------------+-------------+------------------+

-- 比较运算返回 BOOLEAN
SELECT 1 > 0 AS compare_result, 1 = 2 AS equal_result;
-- +-----------------+---------------+
-- | compare_result  | equal_result  |
-- +-----------------+---------------+
-- | true            | false         |
-- +-----------------+---------------+

-- BOOLEAN 的隐式转换：非零数值视为 TRUE，零视为 FALSE
SELECT
    CAST(1 AS BOOLEAN) AS int_true,
    CAST(0 AS BOOLEAN) AS int_false;
-- +-----------+------------+
-- | int_true  | int_false  |
-- +-----------+------------+
-- | true      | false      |
-- +-----------+------------+
```

# 5.复杂类型（Complex Types）
## 5.1 `ARRAY`
- `array<data_type>`，例如：`array<int>`。<br>
- 从0.14版本开始，允许使用负值和非常数表达式。<br>

```sql
-- 建表时指定 ARRAY 类型
CREATE TABLE array_demo (
    id INT,
    scores ARRAY<INT> COMMENT '分数列表',
    tags ARRAY<STRING> COMMENT '标签列表'
);

-- 插入数据（使用 array() 函数构造）
INSERT INTO array_demo VALUES
    (1, array(90, 85, 92, 78), array('math', 'english', 'science')),
    (2, array(100, 88), array('history', 'art')),
    (3, array(), array('music'));

-- 查询
SELECT * FROM array_demo;
-- +-----+------------------+--------------------------------+
-- | id  | scores           | tags                           |
-- +-----+------------------+--------------------------------+
-- | 1   | [90,85,92,78]    | ["math","english","science"]   |
-- | 2   | [100,88]         | ["history","art"]              |
-- | 3   | []               | ["music"]                      |
-- +-----+------------------+--------------------------------+

-- 通过索引访问 ARRAY 元素（索引从 0 开始）
SELECT id, scores[0] AS first_score, tags[1] AS second_tag FROM array_demo;
-- +-----+--------------+-------------+
-- | id  | first_score  | second_tag  |
-- +-----+--------------+-------------+
-- | 1   | 90           | english     |
-- | 2   | 100          | art         |
-- | 3   | NULL         | NULL        |
-- +-----+--------------+-------------+

-- ARRAY 常用函数
SELECT
    SIZE(scores) AS array_size,
    ARRAY_CONTAINS(tags, 'math') AS has_math,
    SORT_ARRAY(scores) AS sorted_scores
FROM array_demo WHERE id = 1;
-- +-------------+-----------+--------------------+
-- | array_size  | has_math  | sorted_scores      |
-- +-------------+-----------+--------------------+
-- | 4           | true      | [78,85,90,92]      |
-- +-------------+-----------+--------------------+

-- 使用 EXPLODE 展开 ARRAY
SELECT id, score
FROM array_demo
LATERAL VIEW EXPLODE(scores) t AS score
WHERE id = 1;
-- +-----+--------+
-- | id  | score  |
-- +-----+--------+
-- | 1   | 90     |
-- | 1   | 85     |
-- | 1   | 92     |
-- | 1   | 78     |
-- +-----+--------+

-- 空数组判断
SELECT * FROM array_demo WHERE SIZE(scores) = 0;
-- +-----+---------+---------+
-- | id  | scores  | tags    |
-- +-----+---------+---------+
-- | 3   | []      | ["music"]|
-- +-----+---------+---------+
```

## 5.2 `MAP`
- `map<key_type, value_type>`，例如：`map<string, int>`。<br>
- 从0.14版本开始，允许使用负值和非常数表达式。<br>

```sql
-- 建表时指定 MAP 类型
CREATE TABLE map_demo (
    id INT,
    properties MAP<STRING, STRING> COMMENT '属性键值对',
    scores_map MAP<STRING, INT> COMMENT '科目分数映射'
);

-- 插入数据（使用 map() 函数构造）
INSERT INTO map_demo VALUES
    (1,
     map('color', 'red', 'size', 'L', 'brand', 'Nike'),
     map('math', 95, 'english', 88, 'science', 92)),
    (2,
     map('color', 'blue', 'size', 'M'),
     map('math', 78, 'history', 85));

-- 查询
SELECT * FROM map_demo;
-- +-----+----------------------------------------------+------------------------------------+
-- | id  | properties                                   | scores_map                         |
-- +-----+----------------------------------------------+------------------------------------+
-- | 1   | {"color":"red","size":"L","brand":"Nike"}    | {"math":95,"english":88,...}       |
-- | 2   | {"color":"blue","size":"M"}                  | {"math":78,"history":85}           |
-- +-----+----------------------------------------------+------------------------------------+

-- 通过 Key 访问 MAP 中的值
SELECT
    id,
    properties['color'] AS color,
    scores_map['math'] AS math_score
FROM map_demo;
-- +-----+--------+-------------+
-- | id  | color  | math_score  |
-- +-----+--------+-------------+
-- | 1   | red    | 95          |
-- | 2   | blue   | 78          |
-- +-----+--------+-------------+

-- MAP 常用函数
SELECT
    SIZE(properties) AS map_size,
    MAP_KEYS(properties) AS all_keys,
    MAP_VALUES(scores_map) AS all_values
FROM map_demo WHERE id = 1;
-- +-----------+--------------------------+----------------+
-- | map_size  | all_keys                 | all_values     |
-- +-----------+--------------------------+----------------+
-- | 3         | ["color","size","brand"] | [95,88,92]     |
-- +-----------+--------------------------+----------------+

-- 使用 EXPLODE 展开 MAP
SELECT id, subject, score
FROM map_demo
LATERAL VIEW EXPLODE(scores_map) t AS subject, score
WHERE id = 1;
-- +-----+----------+--------+
-- | id  | subject  | score  |
-- +-----+----------+--------+
-- | 1   | math     | 95     |
-- | 1   | english  | 88     |
-- | 1   | science  | 92     |
-- +-----+----------+--------+
```

## 5.3 `STRUCT`
- `struct<col_name : data_type [comment col_comment], ...>`，例如：`struct<value: string>`。<br>

```sql
-- 建表时指定 STRUCT 类型
CREATE TABLE struct_demo (
    id INT,
    address STRUCT<city:STRING, street:STRING, zipcode:STRING> COMMENT '地址结构体',
    person STRUCT<name:STRING, age:INT, email:STRING> COMMENT '人员信息'
);

-- 插入数据（使用 named_struct() 函数构造）
INSERT INTO struct_demo VALUES
    (1,
     named_struct('city', 'Beijing', 'street', 'Chang\'an Ave', 'zipcode', '100000'),
     named_struct('name', 'Alice', 'age', 30, 'email', 'alice@example.com')),
    (2,
     named_struct('city', 'Shanghai', 'street', 'Nanjing Rd', 'zipcode', '200000'),
     named_struct('name', 'Bob', 'age', 25, 'email', 'bob@example.com'));

-- 查询
SELECT * FROM struct_demo;
-- +-----+---------------------------------------------------+---------------------------------------------------+
-- | id  | address                                           | person                                            |
-- +-----+---------------------------------------------------+---------------------------------------------------+
-- | 1   | {"city":"Beijing","street":"Chang'an Ave",...}    | {"name":"Alice","age":30,"email":"alice@..."}     |
-- | 2   | {"city":"Shanghai","street":"Nanjing Rd",...}    | {"name":"Bob","age":25,"email":"bob@..."}         |
-- +-----+---------------------------------------------------+---------------------------------------------------+

-- 通过点号访问 STRUCT 中的字段
SELECT
    id,
    address.city AS city,
    address.street AS street,
    person.name AS name,
    person.age AS age
FROM struct_demo;
-- +-----+-----------+---------------+--------+------+
-- | id  | city      | street        | name   | age  |
-- +-----+-----------+---------------+--------+------+
-- | 1   | Beijing   | Chang'an Ave  | Alice  | 30   |
-- | 2   | Shanghai  | Nanjing Rd    | Bob    | 25   |
-- +-----+-----------+---------------+--------+------+

-- 嵌套 STRUCT 示例
CREATE TABLE nested_struct_demo (
    id INT,
    order_info STRUCT<
        order_id:STRING,
        customer:STRUCT<name:STRING, phone:STRING>,
        items:ARRAY<STRUCT<item_name:STRING, price:DOUBLE>>
    >
);

INSERT INTO nested_struct_demo VALUES (
    1,
    named_struct(
        'order_id', 'ORD-001',
        'customer', named_struct('name', 'Charlie', 'phone', '13800138000'),
        'items', array(
            named_struct('item_name', 'Laptop', 'price', 5999.00),
            named_struct('item_name', 'Mouse', 'price', 99.00)
        )
    )
);

-- 多层字段访问
SELECT
    id,
    order_info.order_id AS order_id,
    order_info.customer.name AS customer_name,
    order_info.items[0].item_name AS first_item,
    order_info.items[0].price AS first_item_price
FROM nested_struct_demo;
-- +-----+-----------+----------------+-------------+------------------+
-- | id  | order_id  | customer_name  | first_item  | first_item_price |
-- +-----+-----------+----------------+-------------+------------------+
-- | 1   | ORD-001   | Charlie        | Laptop      | 5999.0           |
-- +-----+-----------+----------------+-------------+------------------+
```

## 5.4 `UNIONTYPE`
- 从0.7.0版本开始使用。<br>
- `uniontype<data_type, data_type, ...>`，例如：`uniontype<int, string>`。<br>

```sql
-- 建表时指定 UNIONTYPE 类型
CREATE TABLE union_demo (
    id INT,
    value UNIONTYPE<INT, STRING, DOUBLE> COMMENT '可以是 INT、STRING 或 DOUBLE'
);

-- 插入数据（使用 create_union() 函数构造）
-- create_union(tag, value): tag 为类型索引（从 0 开始）
INSERT INTO union_demo VALUES
    (1, create_union(0, 100, 3.14, 'hello')),           -- tag=0, INT 类型: 100
    (2, create_union(1, 200, 3.14, 'hello')),           -- tag=1, STRING 类型: 3.14 被转为字符串
    (3, create_union(2, 200, 3.14, 'hello'));           -- tag=2, DOUBLE 类型: 3.14

-- 查询
SELECT * FROM union_demo;
-- +-----+------------+
-- | id  | value      |
-- +-----+------------+
-- | 1   | {0:100}    |
-- | 2   | {1:"3.14"} |
-- | 3   | {2:3.14}   |
-- +-----+------------+

-- UNIONTYPE 实际中较少使用，更推荐使用 STRUCT 替代
-- 如果确实需要多类型字段，可以考虑以下替代方案：

-- 替代方案1: 使用多个单独列 + CASE WHEN
CREATE TABLE better_union_demo (
    id INT,
    int_val INT,
    str_val STRING,
    dbl_val DOUBLE,
    active_type STRING COMMENT '标识当前有效类型: INT/STRING/DOUBLE'
);

INSERT INTO better_union_demo VALUES
    (1, 100, NULL, NULL, 'INT'),
    (2, NULL, 'hello', NULL, 'STRING');

-- 替代方案2: 使用 STRUCT 包装
CREATE TABLE struct_union_demo (
    id INT,
    value STRUCT<type_tag:INT, int_val:INT, str_val:STRING, dbl_val:DOUBLE>
);
```
